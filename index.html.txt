    <script>
        // CONFIGURAÇÃO DO SUPABASE
        const SUPABASE_URL = 'https://utwhgyfmuaalacbtvqxx.supabase.co';
        const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6InV0d2hneWZtdWFhbGFjYnR2cXh4Iiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODk1OTYxMzQsImV4cCI6MjEwNTE3MjEzNH0.Bm-SuGODW8vMy54ZEeHTaGOm8GnCvxA3B-vF6pt6HlM';
        
        const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);
        let usuarioAtual = null;

        async function verificarSessao() {
            const { data: { session } } = await supabaseClient.auth.getSession();
            if (session) {
                usuarioAtual = session.user;
                document.getElementById('usuarioLogado').innerText = usuarioAtual.email;
                document.getElementById('telaLogin').classList.add('hidden');
                document.getElementById('appPrincipal').classList.remove('hidden');
                carregarDados();
                escutarTempoReal();
            }
        }
        verificarSessao();

        async function fazerLogin(e) {
            e.preventDefault();
            const email = document.getElementById('loginEmail').value;
            const senha = document.getElementById('loginSenha').value;
            const erroDiv = document.getElementById('erroLogin');

            const { data, error } = await supabaseClient.auth.signInWithPassword({ email, senha });
            if (error) {
                erroDiv.innerText = "Erro ao entrar: " + error.message;
                erroDiv.classList.remove('hidden');
            } else {
                window.location.reload();
            }
        }

        async function fazerLogout() {
            await supabaseClient.auth.signOut();
            window.location.reload();
        }

        function mudarAba(aba) {
            const secAbordagens = document.getElementById('secaoAbordagens');
            const secPlantoes = document.getElementById('secaoPlantoes');
            const btnAbordagens = document.getElementById('btnAbordagens');
            const btnPlantoes = document.getElementById('btnPlantoes');

            if (aba === 'abordagens') {
                secAbordagens.classList.remove('hidden');
                secPlantoes.classList.add('hidden');
                btnAbordagens.className = "bg-white text-blue-600 px-4 py-2 rounded font-semibold transition shadow";
                btnPlantoes.className = "bg-blue-700 text-white px-4 py-2 rounded font-semibold transition shadow";
            } else {
                secPlantoes.classList.remove('hidden');
                secAbordagens.classList.add('hidden');
                btnPlantoes.className = "bg-white text-blue-600 px-4 py-2 rounded font-semibold transition shadow";
                btnAbordagens.className = "bg-blue-700 text-white px-4 py-2 rounded font-semibold transition shadow";
            }
        }

        async function carregarDados() {
            carregarAbordagens();
            carregarPlantoes();
        }

        async function carregarAbordagens() {
            const { data, error } = await supabaseClient.from('abordagens').select('*').order('created_at', { ascending: false });
            if (!error) renderizarTabelaAbordagens(data);
        }

        async function carregarPlantoes() {
            const { data, error } = await supabaseClient.from('plantoes').select('*').order('data', { ascending: true });
            if (!error) renderizarTabelaPlantoes(data);
        }

        async function salvarAbordagem(e) {
            e.preventDefault();
            const nova = {
                nome_cliente: document.getElementById('nomeCliente').value,
                telefone: document.getElementById('telCliente').value,
                bloco_apto: document.getElementById('blocoApto').value,
                operadora: document.getElementById('operadoraAtual').value,
                consultor_email: usuarioAtual.email,
                user_id: usuarioAtual.id
            };

            const { error } = await supabaseClient.from('abordagens').insert([nova]);
            if (error) {
                alert('Erro ao salvar: ' + error.message);
            } else {
                document.getElementById('formAbordagem').reset();
            }
        }

        async function salvarPlantao(e) {
            e.preventDefault();
            const dataFiltro = document.getElementById('plantaoData').value;
            const turno = document.getElementById('plantaoTurno').value;

            const novo = {
                data: dataFiltro,
                turno: turno,
                consultor_email: usuarioAtual.email,
                user_id: usuarioAtual.id
            };

            const { error } = await supabaseClient.from('plantoes').insert([novo]);
            if (error) {
                alert('Erro ao agendar plantão: ' + error.message);
            } else {
                document.getElementById('formPlantao').reset();
            }
        }

        function renderizarTabelaAbordagens(lista) {
            const tbody = document.getElementById('tabelaAbordagens');
            const contador = document.getElementById('contadorAbordagens');
            tbody.innerHTML = '';
            contador.innerText = `${lista.length} registros`;

            if (lista.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="p-4 text-center text-gray-400">Nenhuma abordagem registrada.</td></tr>`;
                return;
            }

            lista.forEach(item => {
                const dataFormatada = new Date(item.created_at).toLocaleString('pt-BR', {dateStyle: 'short', timeStyle: 'short'});
                const podeExcluir = item.user_id === usuarioAtual.id;
                
                tbody.innerHTML += `
                    <tr class="hover:bg-gray-50">
                        <td class="p-3 text-xs text-gray-500">${dataFormatada}</td>
                        <td class="p-3 font-medium text-gray-800">${item.nome_cliente}</td>
                        <td class="p-3"><a href="https://wa.me/55${item.telefone.replace(/\D/g,'')}" target="_blank" class="text-green-600 hover:underline"><i class="fa-brands fa-whatsapp"></i> ${item.telefone}</a></td>
                        <td class="p-3">${item.bloco_apto}</td>
                        <td class="p-3"><span class="bg-gray-200 text-gray-700 px-2 py-0.5 rounded text-xs font-semibold">${item.operadora}</span></td>
                        <td class="p-3 text-gray-600">${item.consultor_email}</td>
                        <td class="p-3 text-center">
                            ${podeExcluir ? `<button onclick="excluirRegistro('abordagens', '${item.id}')" class="text-red-500 hover:text-red-700"><i class="fa-solid fa-trash"></i></button>` : '<span class="text-gray-300">-</span>'}
                        </td>
                    </tr>
                `;
            });
        }

        function renderizarTabelaPlantoes(lista) {
            const tbody = document.getElementById('tabelaPlantoes');
            tbody.innerHTML = '';

            if (lista.length === 0) {
                tbody.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-gray-400">Nenhum plantão agendado.</td></tr>`;
                return;
            }

            lista.forEach(item => {
                const dataBR = item.data.split('-').reverse().join('/');
                const podeExcluir = item.user_id === usuarioAtual.id;

                tbody.innerHTML += `
                    <tr class="hover:bg-gray-50">
                        <td class="p-3 font-medium text-gray-800">${dataBR}</td>
                        <td class="p-3">${item.consultor_email}</td>
                        <td class="p-3"><span class="bg-blue-100 text-blue-800 px-2 py-0.5 rounded text-xs font-semibold">${item.turno}</span></td>
                        <td class="p-3 text-center">
                            ${podeExcluir ? `<button onclick="excluirRegistro('plantoes', '${item.id}')" class="text-red-500 hover:text-red-700"><i class="fa-solid fa-trash"></i></button>` : '<span class="text-gray-300">-</span>'}
                        </td>
                    </tr>
                `;
            });
        }

        async function excluirRegistro(tabela, id) {
            if (confirm('Deseja realmente excluir este item?')) {
                const { error } = await supabaseClient.from(tabela).delete().eq('id', id);
                if (error) alert('Erro ao excluir: ' + error.message);
            }
        }

        function escutarTempoReal() {
            supabaseClient.channel('public-db-changes')
                .on('postgres_changes', { event: '*', schema: 'public', table: 'abordagens' }, payload => {
                    carregarAbordagens();
                })
                .on('postgres_changes', { event: '*', schema: 'public', table: 'plantoes' }, payload => {
                    carregarPlantoes();
                })
                .subscribe();
        }
    </script>
