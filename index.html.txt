    <script>
        function mudarAba(aba) {
            const secAbordagens = document.getElementById('secaoAbordagens');
            const secPlantoes = document.getElementById('secaoPlantoes');
            const btnAbordagens = document.getElementById('btnAbordagens');
            const btnPlantoes = document.getElementById('btnPlantoes');

            if (aba === 'abordagens') {
                secAbordagens.classList.remove('hidden');
                secPlantoes.classList.add('hidden');
                btnAbordagens.className = "bg-white text-blue-600 px-4 py-2 rounded font-semibold transition shadow";
                btnPlantoes.className = "bg-blue-700 text-white px-4 py-2 rounded font-semibold transition shadow";
            } else {
                secPlantoes.classList.remove('hidden');
                secAbordagens.classList.add('hidden');
                btnPlantoes.className = "bg-white text-blue-600 px-4 py-2 rounded font-semibold transition shadow";
                btnAbordagens.className = "bg-blue-700 text-white px-4 py-2 rounded font-semibold transition shadow";
            }
        }

        let abordagens = JSON.parse(localStorage.getItem('abordagens')) || [];
        let plantoes = JSON.parse(localStorage.getItem('plantoes')) || [];

        function salvarAbordagem(e) {
            e.preventDefault();
            const novaAbordagem = {
                id: Date.now(),
                data: new Date().toLocaleDateString('pt-BR') + ' ' + new Date().toLocaleTimeString('pt-BR', {hour: '2-digit', minute:'2-digit'}),
                nome: document.getElementById('nomeCliente').value,
                telefone: document.getElementById('telCliente').value,
                bloco: document.getElementById('blocoApto').value,
                operadora: document.getElementById('operadoraAtual').value,
                consultor: document.getElementById('consultorNome').value
            };

            abordagens.unshift(novaAbordagem);
            localStorage.setItem('abordagens', JSON.stringify(abordagens));
            document.getElementById('formAbordagem').reset();
            renderizarAbordagens();
        }

        function renderizarAbordagens() {
            const tbody = document.getElementById('tabelaAbordagens');
            const contador = document.getElementById('contadorAbordagens');
            tbody.innerHTML = '';
            contador.innerText = `${abordagens.length} registros`;

            if (abordagens.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="p-4 text-center text-gray-400">Nenhuma abordagem registrada ainda.</td></tr>`;
                return;
            }

            abordagens.forEach(item => {
                tbody.innerHTML += `
                    <tr class="hover:bg-gray-50">
                        <td class="p-3 text-xs text-gray-500">${item.data}</td>
                        <td class="p-3 font-medium text-gray-800">${item.nome}</td>
                        <td class="p-3"><a href="https://wa.me/55${item.telefone.replace(/\D/g,'')}" target="_blank" class="text-green-600 hover:underline"><i class="fa-brands fa-whatsapp"></i> ${item.telefone}</a></td>
                        <td class="p-3">${item.bloco}</td>
                        <td class="p-3"><span class="bg-gray-200 text-gray-700 px-2 py-0.5 rounded text-xs font-semibold">${item.operadora}</span></td>
                        <td class="p-3 text-gray-600">${item.consultor}</td>
                        <td class="p-3 text-center">
                            <button onclick="excluirAbordagem(${item.id})" class="text-red-500 hover:text-red-700"><i class="fa-solid fa-trash"></i></button>
                        </td>
                    </tr>
                `;
            });
        }

        function excluirAbordagem(id) {
            if (confirm('Deseja realmente excluir este registro?')) {
                abordagens = abordagens.filter(i => i.id !== id);
                localStorage.setItem('abordagens', JSON.stringify(abordagens));
                renderizarAbordagens();
            }
        }

        function salvarPlantao(e) {
            e.preventDefault();
            const novoPlantao = {
                id: Date.now(),
                consultor: document.getElementById('plantaoConsultor').value,
                data: document.getElementById('plantaoData').value.split('-').reverse().join('/'),
                turno: document.getElementById('plantaoTurno').value
            };

            plantoes.push(novoPlantao);
            localStorage.setItem('plantoes', JSON.stringify(plantoes));
            document.getElementById('formPlantao').reset();
            renderizarPlantoes();
        }

        function renderizarPlantoes() {
            const tbody = document.getElementById('tabelaPlantoes');
            tbody.innerHTML = '';

            if (plantoes.length === 0) {
                tbody.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-gray-400">Nenhum plantão agendado.</td></tr>`;
                return;
            }

            plantoes.push // wait, error check syntax fixed below
            plantoes.forEach(item => {
                tbody.innerHTML += `
                    <tr class="hover:bg-gray-50">
                        <td class="p-3 font-medium text-gray-800">${item.data}</td>
                        <td class="p-3">${item.consultor}</td>
                        <td class="p-3"><span class="bg-blue-100 text-blue-800 px-2 py-0.5 rounded text-xs font-semibold">${item.turno}</span></td>
                        <td class="p-3 text-center">
                            <button onclick="excluirPlantao(${item.id})" class="text-red-500 hover:text-red-700"><i class="fa-solid fa-trash"></i></button>
                        </td>
                    </tr>
                `;
            });
        }

        function excluirPlantao(id) {
            if (confirm('Deseja realmente cancelar este plantão?')) {
                plantoes = plantoes.filter(i => i.id !== id);
                localStorage.setItem('plantoes', JSON.stringify(plantoes));
                renderizarPlantoes();
            }
        }

        renderizarAbordagens();
        renderizarPlantoes();
    </script>
